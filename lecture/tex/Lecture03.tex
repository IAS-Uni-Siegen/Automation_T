\begin{frame}{Overview of Signal Processing and Calibration}
\small
\textbf{Goal:} Convert a physical quantity into a usable electrical signal.

\begin{itemize}
  \item Sensors generate low-level, often non-linear, analog outputs.
  \item Signal processing and conditioning amplify, filter, linearize, and standardize them.
  \item Calibration ensures accuracy and removes offset, drift, and disturbances.
\end{itemize}

\textbf{Key challenges:}
\begin{enumerate}
  \item Very small signal levels $\rightarrow$ require amplification.
  \item Influence of disturbance variables (e.g. temperature).
  \item Nonlinear characteristic curves $\rightarrow$ need linearization and filtering.
\end{enumerate}
\end{frame}

\begin{frame}{Typical Measurement Chain}
\centering

\vspace{4pt}
\small
\textbf{Process:}
\begin{enumerate}
  \item Sensor converts the stimulus.
  \item Signal conditioning amplifies and linearizes.
  \item ADC digitizes the signal.
  \item Microcontroller processes, controls, and communicates.
  \item DAC or actuator output closes the control loop.
\end{enumerate}
\end{frame}

\begin{frame}{Signal Processing: Basic Types}
\small
\textbf{Physical effects generating the sensor signal:}
\begin{itemize}
  \item \textbf{Charge / Voltage generation:} piezoelectric, thermoelectric, photoelectric.
  \item \textbf{Resistance or Conductance change:} strain gauge, thermistor, FET-based sensors.
  \item \textbf{Capacitance or Inductance change:} displacement, proximity, humidity sensors.
\end{itemize}

\textbf{Output types:}
\begin{itemize}
  \item Binary (on/off)
  \item Pulse / frequency encoded
  \item Analog continuous signals
\end{itemize}
\end{frame}


\begin{frame}{Analog (Discrete) Signal Conditioning}
\small
\textbf{Purpose:} Boost and shape low-level analog signals.

\textbf{Features:}
\begin{itemize}
  \item Low-noise amplification (e.g., differential bridge amplifiers).
  \item Offset and gain trimming via resistors.
  \item Filtering of unwanted frequency components.
\end{itemize}

\begin{frame}{System Circuits for Signal Conditioning}
\small
\textbf{Motivation:} Integrate signal processing tasks in ICs for compactness and reliability.

\begin{itemize}
  \item Use of ICs like \textbf{XTR108} (Texas Instruments) for RTD/bridge-to-current loop conversion.
  \item On-chip features:
  \begin{itemize}
    \item Programmable gain amplifier (PGA)
    \item Multiplexer for multiple inputs
    \item Linearization and excitation current source
    \item EEPROM for calibration constants
  \end{itemize}
  \item Standard 4–20 mA industrial output.
\end{itemize}
\end{frame}



\begin{frame}{Signal Conditioning with ASICs and Microcontrollers}
\small
\textbf{ASICs (Application-Specific ICs):}
\begin{itemize}
  \item Tailored to specific sensors (pressure, displacement, etc.)
  \item Contain analog and digital processing, temperature compensation, EEPROM storage.
\end{itemize}

\textbf{Microcontroller-based processing:}
\begin{itemize}
  \item Integrates ADC, DAC, or PWM unit.
  \item Enables flexible digital linearization and compensation.
  \item Supports communication interfaces (SPI, I$^2$C, UART).
\end{itemize}

\textbf{Example:} MSP430 µC with sensor bridge input and display output.
\end{frame}

\begin{frame}{Sensor Calibration: Concepts}
\small
\textbf{Calibration:} Documentation of deviation between measured and true values.

\textbf{Adjustment:} Setting the sensor output to a known standard point (setpoint).

\textbf{Compensation techniques:}
\begin{itemize}
  \item \textbf{Passive compensation:} Using resistive bridges (trim R1–R4 for zero and slope).
  \item \textbf{Analog compensation:} Op-amp circuits adjusting offset/gain.
  \item \textbf{Digital compensation:} Performed in ASICs or µCs via software.
\end{itemize}
\end{frame}

\begin{frame}{Analog and Digital Adjustment Methods}
\small
\textbf{Analog signal processing (ASIC SCA2096):}
\begin{itemize}
  \item Uses D/A converters for gain and offset correction.
  \item Temperature influence compensated via secondary correction path.
  \item Three-step calibration: high, room, and low temperature.
\end{itemize}

\textbf{Digital signal processing:}
\begin{itemize}
  \item Offset/gain correction and linearization in software.
  \item Employs high-resolution ADC + digital correction function.
  \item Uses either arithmetic equations or lookup tables.
\end{itemize}
\end{frame}


\begin{frame}{Digital Correction Strategies}
\small
\textbf{Arithmetic method:}
\begin{itemize}
  \item Uses polynomial equations with coefficients determined during calibration.
  \item Example: $Y = Z_p + \text{Offset}(Z_{T1}) / \text{Gain}(Z_{T1})$.
\end{itemize}

\textbf{Characteristic curve interpolation:}
\begin{itemize}
  \item Interpolates between stored calibration points.
  \item Linear interpolation sufficient for dense data; higher orders for complex curves.
\end{itemize}

\textbf{Lookup Table (LUT):}
\begin{itemize}
  \item Fastest method; stores precomputed outputs.
  \item High memory consumption, but ideal for embedded real-time applications.
\end{itemize}
\end{frame}

\begin{frame}{Emergence of Smart Sensors}
\small
\textbf{Smart sensors = Sensor + Conditioning + Processing + Communication}

\begin{itemize}
  \item Integrate:
    \begin{itemize}
      \item Precision amplifier and ADC
      \item Microcontroller or DSP
      \item Digital interface (SPI, I$^2$C, fieldbus)
    \end{itemize}
  \item Provide self-calibration, self-diagnostics, and standard digital output.
  \item Enable plug-and-play interoperability in industrial networks.
\end{itemize}
