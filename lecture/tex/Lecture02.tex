\subsection{Introduction to Sensor Systems}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Introduction to Sensor Systems %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Introduction to Sensor Systems}
    \begin{itemize}
        \item Sensors are essential to observe and influence physical, chemical, or biological processes.
        \item They enable control of systems by providing information about the \textbf{current state}.
        \item Without sensors, automated or intelligent systems cannot perceive their environment.
        \item Sensor systems combine physical measurement principles with electronic evaluation and control.
    \end{itemize}
    \vspace{2mm}
    \textbf{Key idea:} To control a process, one must first \textit{measure}, then \textit{evaluate}, and finally \textit{act}.
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Definition and Mode of Operation %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Definition and Mode of Operation}
    \begin{block}{Definition}
        The term \textbf{sensor} originates from the Latin \textit{sensus} (sense, feeler).  
        A sensor is a device that converts a physical, chemical, or biological quantity into an \textbf{electrical output signal}.
    \end{block}
    \begin{itemize}
        \item Sensors are used for quantitative and qualitative measurements of various parameters (e.g., temperature, pressure, acceleration, chemical concentration).
        \item Each sensor system consists of two essential parts:
        \begin{enumerate}
            \item \textbf{Sensor element:} converts non-electrical input variables (e.g., pressure, light) into an electrical signal.
            \item \textbf{Evaluation electronics:} amplifies, filters, or digitizes the signal for control or monitoring purposes.
        \end{enumerate}
    \end{itemize}
    \vspace{-0.5cm}
    \begin{figure}
        \centering
        \includegraphics[width=0.5\linewidth]{fig/lec02/sensor_intro.pdf}
        \caption{Operating principle of a sensor system (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
    \end{figure}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Operating Principle of Sensors %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Operating Principle of Sensors}
    \begin{itemize}
        \item The sensor receives \textbf{input variables} (physical or non-electrical quantities).
        \item It produces an \textbf{electrical output signal} proportional to the input.
        \item External \textbf{disturbance variables} (temperature, vibration, noise) can affect accuracy.
        \item Modern sensors compensate disturbances using embedded microprocessors and calibration algorithms.
        \item The combination of sensing element and processing electronics is known as a \textbf{smart sensor}.
    \end{itemize}
    \begin{block}{Smart Sensor}
        A smart sensor integrates the sensing element and electronic evaluation in a single compact unit, capable of self-calibration and digital communication.
    \end{block}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Classification of Sensors %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Classification of Sensors}
    \textbf{1. Based on Power Source}
    \begin{itemize}
        \item \textbf{Active sensors:} Generate an electrical signal directly from the measured variable (e.g., piezoelectric, thermoelectric).
        \item \textbf{Passive sensors:} Require external auxiliary power to produce an output (e.g., resistive, capacitive, inductive, ultrasonic).
    \end{itemize}
    \vspace{2mm}
    \textbf{2. Based on Measurement Principle}
    \begin{itemize}
        \item Mechanical, thermal, magnetic, optical, chemical, biological, etc.
    \end{itemize}
    \vspace{2mm}
    \textbf{3. Based on Nature of Output}
    \begin{itemize}
        \item \textbf{Analog sensors:} Continuous output (e.g., thermistor).
        \item \textbf{Digital sensors:} Include A/D conversion and provide discrete signals or data streams.
    \end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Classification of Sensors %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Digital and Virtual Sensors}
    \textbf{Digital Sensors}
    \begin{itemize}
        \item Integrate \textbf{analog-to-digital converters (ADCs)} within the evaluation electronics.
        \item Output is a \textbf{digital signal} directly usable by controllers or microprocessors.
        \item Example: Digital accelerometer (I\textsuperscript{2}C/SPI interface).
    \end{itemize}
    \vspace{3mm}
    \textbf{Virtual Sensors}
    \begin{itemize}
        \item Derive the desired variable \textbf{computationally} using software models.
        \item Based on mathematical models or empirical correlations.
        \item Used when:
        \begin{itemize}
            \item Real sensors are expensive or prone to wear (e.g., nuclear plants, harsh environments).
            \item Quantities can be inferred indirectly (e.g., estimated torque from current and speed).
        \end{itemize}
    \end{itemize}
    \begin{block}{Example}
        In automotive systems, a virtual tire pressure sensor can estimate pressure from wheel-speed variations and temperature.
    \end{block}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Summary and Discussion %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Summary and Discussion}
    \begin{itemize}
        \item Sensors are the foundation of automation and control.
        \item Each sensor has a sensing element and signal-conditioning electronics.
        \item Classification helps select the right sensor for a given task:
            \begin{itemize}
                \item Active vs Passive
                \item Analog vs Digital
                \item Real vs Virtual
            \end{itemize}
        \item Emerging trends: miniaturization, MEMS, IoT-ready smart sensors.
    \end{itemize}
    \vspace{3mm}
    \textbf{Classroom Question:}
    \begin{quote}
        Why might a virtual sensor be preferred over a physical sensor in a battery management or air suspension system?
    \end{quote}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% What is the Piezoelectric Effect? %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{What is the Piezoelectric Effect?}
    \textbf{Direct effect:} When certain solids are \emph{deformed} by force/pressure, internal charge centers shift $\Rightarrow$ an \textbf{electrical voltage} appears at the surfaces (electric polarization $P$).\\[2mm]
    \textbf{Inverse effect:} Applied voltage causes a \emph{mechanical strain/oscillation}.\\[2mm]
    \textbf{Key ideas from the text}
    \begin{itemize}
        \item Polarization $P$ depends on direction of force $\mathbf{F}$ and crystal axes.
        \item Only \emph{changes} in force generate charge (static force alone produces no output).
        \item Effect is temperature dependent and can be nonlinear at high fields.
    \end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Principle of Operation (Direct) and Modes %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Principle of Operation (Direct) and Modes}
    \begin{columns}[T,onlytextwidth]
        \column{0.65\textwidth}
            \textbf{Operating chain}\\
            Force/pressure $\rightarrow$ charge displacement $\rightarrow$ surface charge $\rightarrow$ measurable voltage.\\[2mm]
            \textbf{Common modes:}
            \begin{itemize}
                \item \textbf{Longitudinal} / thickness oscillation (stress $\parallel$ pol.\ dir.).
                \item \textbf{Transverse} strain element (stress $\perp$ pol.\ dir.).
                \item \textbf{Bimorph} (two layers bend under field/force).
            \end{itemize}
        \column{0.35\textwidth}
            \centering
            \begin{figure}
                \includegraphics[width=0.4\linewidth]{fig/lec02/piezo_3.pdf}\\
                \caption{Principle: force $\rightarrow$ voltage (longitudinal) (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
    \end{columns}
    \centering
    \begin{figure}
        \includegraphics[width=0.5\linewidth]{fig/lec02/piezo_1.pdf}\\
        \caption{The three principles of the piezoelectric effect (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
    \end{figure}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Coordinate Description and Basic Relations %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Coordinate Description and Basic Relations}
    \begin{columns}[T,onlytextwidth]
        \column{0.48\textwidth}
            \textbf{Coordinate system}\\
            Indices $i\in\{1,2,3\}$ denote force/field directions, $j\in\{1,2,3\}$ surface normals.\\[2mm]
            \textbf{Constitutive relations}
            \begin{equation}
                Q = d_{kl}\,F, \qquad
                P = e\,\sigma, \qquad
                \sigma_{ij} = \frac{F_i}{A_j}
            \end{equation}
            \textit{$d_{kl}$: piezoelectric charge coefficient; $e$: piezoelectric elongation coefficient.}
        \column{0.48\textwidth}
            \centering
            \begin{figure}
                \includegraphics[width=\linewidth]{fig/lec02/piezo_2.pdf}\\
                \caption{Axes and polarization direction (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
    \end{columns}
    \vspace{2mm}
    \textbf{Measurement note:} Because $Q \propto \dot{F}$, output is often \emph{current-like}; integrating current yields charge proportional to force change.
\end{frame}

\begin{frame}{Design and Measurement Tips}
    \begin{itemize}
        \item \textbf{Dynamic quantity:} best for varying loads; use charge or IEPE amplifiers for low-frequency work.
        \item \textbf{Calibration:} temperature and preload affect sensitivity; account for nonlinearity at high fields.

\subsection{Resistive and Piezoresistive Effect}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Operating Principle (Resistive Effect) %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Operating Principle (Resistive Effect)}
    \textbf{Specific resistivity $\rho$} of a metal is (to first order) \emph{independent of voltage}.\\
    Electrical resistance of a conductor of length $l_0$ and cross-section $A_0$:
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{equation}
                R=\rho\frac{l_0}{A_0}
            \end{equation}
            Implications:
            \begin{itemize}
                \item $R \propto l$ (stretched $\Rightarrow$ longer) \quad
                    $R \propto \tfrac{1}{A}$ (stretched $\Rightarrow$ thinner).
                \item In the elastic range, \textbf{tension} increases $R$; \textbf{compression} decreases $R$.
            \end{itemize}
        \end{column}
        \begin{column}{0.40\textwidth}
            \centering
            \begin{figure}
                \includegraphics[width=\linewidth]{fig/lec02/res_1.pdf} \\
                \caption{Conductor under mechanical stress (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
        \end{column}
    \end{columns}
    \medskip
    \emph{Mechanical stress} $\sigma=\dfrac{F}{A}$, strain $\varepsilon=\dfrac{\Delta l}{l}$, Hooke: $\sigma=E\varepsilon$.
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% From Strain to Resistance Change %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{From Strain to Resistance Change}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            Under axial load, the effective resistance becomes
            \begin{equation}
                R_F=\rho\,\frac{l_F}{A_F}
            \end{equation}
            The relative change is (geometric effect):
            \begin{equation}
                \frac{\Delta R}{R}=k\,\varepsilon
            \end{equation}
        \end{column}
        \begin{column}{0.40\textwidth}
            \centering
            \begin{figure}
                \includegraphics[width=\linewidth]{fig/lec02/res_1.pdf} \\
                \caption{Conductor under mechanical stress (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
        \end{column}
    \end{columns}
    \textbf{$k$ (gauge factor)} $\approx 2$ for metals (lengthening + Poisson contraction).\\
    For \textbf{semiconductors}, $\rho$ also changes with stress (carrier mobility $\Rightarrow$ \textbf{piezoresistive effect}), giving much larger $k$.
    % \includegraphics[width=0.42\linewidth]{f851366a-...png} % optional
\end{frame}

\begin{frame}{Foil Strain Gauges (SGs)}
    \textbf{Construction:}
    \begin{itemize}
        \item Meandered metallic grid (e.g., Constantan, Karma) on polyimide carrier; protective film.
        \item Bonded to the test surface by spot welding (rare) or adhesives (cyanoacrylates, epoxies).
    \end{itemize}
    \textbf{Typical numbers:}
    \begin{itemize}
        \item Commercial SG: $R\approx \SI{120}{\ohm}$; for $\varepsilon=10^{-3}$ and $k\approx2$, $\Delta R\approx \SI{0.24}{\ohm}$.
        \item Small $\Delta R$ \(\Rightarrow\) use \textbf{Wheatstone bridge} and low-noise instrumentation.
    \end{itemize}

\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Wheatstone Bridge Readout %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Wheatstone Bridge Readout}
    \begin{columns}[T,onlytextwidth]
        \column{0.56\textwidth}
            \textbf{General bridge:}
            Four arms $R_1..R_4$, supply $U_B$, output $U_A$.\\[1mm]
            \begin{equation}
                \frac{U_A}{U_B}=\frac{R_1}{R_1+R_2}-\frac{R_4}{R_3+R_4}
            \end{equation}
            For small changes $\Delta R_i \ll R_i$:
            \begin{align*}
                \frac{U_A}{U_B}&=\frac{1}{4}\!\left(
                \frac{\Delta R_1}{R_1}-\frac{\Delta R_2}{R_2}
                +\frac{\Delta R_3}{R_3}- \frac{\Delta R_4}{R_4}\right) \\
                &=\frac{k}{4}\left(\varepsilon_1-\varepsilon_2+\varepsilon_3-\varepsilon_4\right)
            \end{align*}
        \column{0.42\textwidth}
            \centering
            \begin{figure}
                \includegraphics[width=\linewidth]{fig/lec02/res_2.pdf}
                \caption{Quarter,  half or full bridges: combine gauges to boost signal, cancel temperature, and measure bending, axial, torsion selectively (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
    \end{columns}
\end{frame}

%% Magnetoresistive Effect: Principle of Operation %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Magnetoresistive Effect: Principle of Operation}
    \small
    \textbf{Concept:} When a magnetic field is applied to a conducting material through which current flows, the electrical resistance changes. This is known as the \textbf{magnetoresistive effect (MR)}.\\
    \medskip
    \textbf{Mathematical Definition:}
    \begin{equation}
        \frac{\Delta R}{R} = \frac{R(H) - R(0)}{R(0)}
    \end{equation}
    where $R(H)$ is the resistance in magnetic field $H$ and $R(0)$ is resistance without magnetic field.
    \medskip
    \textbf{Physical Principle:}
    \begin{itemize}
        \item In presence of $H$, charge carriers experience a Lorentz force:
        \begin{equation}
            F_L = Q (\mathbf{v} \times \mathbf{B})
        \end{equation}
        \item This deflects electrons, increasing their path length and hence resistance.
        \item Effect depends on material type (magnetic or non-magnetic) and electron mobility.
    \end{itemize}
\end{frame}

\begin{frame}{Classification of Magnetoresistive Effects (2)}
    \textbf{Overview of Magnetoresistive (MR) Effects:}
    \begin{columns}[T]
        \column{0.4\textwidth}
            \textbf{Non-magnetic materials (MR):}
            \begin{itemize} \small
                \item \textbf{OMR - Ordinary Magnetoresistance:}
                Caused by Lorentz-force deflection of charge carriers in a magnetic field.\\
                Effect is small ($<1\%$) in most metals, but large in semimetals like Bi (up to 100\%).
                \item \textbf{EMR - Extraordinary Magnetoresistance:}
                Observed in hybrid metal/semiconductor structures (e.g., InSb).\\
                Large resistance change due to field-induced current redistribution.
            \end{itemize}
        \column{0.6\textwidth}
            \textbf{Magnetic materials (XMR):}
            \begin{itemize} \small
                \item \textbf{AMR - Anisotropic MR:}
                Change of 3-4 \% depending on angle between current and magnetization. Caused by spin-orbit interaction in ferromagnets (e.g., NiFe alloys).
                \item \textbf{GMR - Giant MR:}
                6-100 \% change in multilayer structures (FM/NM/FM). Due to spin-dependent electron scattering at interfaces.
                \item \textbf{CMR - Colossal MR:}
                $>$100 \% change in perovskite manganites (e.g., La$_{1-x}$Sr$_x$MnO$_3$).\\
                Linked to double-exchange interaction and phase transitions.
                \item \textbf{TMR - Tunnel MR:}
                Up to 50 \% change in magnetic tunnel junctions (FM/insulator/FM).\\
                Arises from spin-polarized tunneling; depends on relative magnetization angle.
            \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Magnetostrictive Effect: Principle of Operation (1)}
    \begin{columns}
        \column{0.45\textwidth}
            \textbf{Concept:}
            When a magnetic field is applied to a ferromagnetic material, its length changes by $\Delta l$.  
            This phenomenon is called \textbf{magnetostriction}.
            \medskip
            \textbf{Key Relations:}
            \begin{equation}
                \varepsilon = \frac{\Delta l}{l} = \kappa^* \frac{H}{E}
            \end{equation}
            where $\kappa^*$ = magnetostriction constant, $H$ = magnetic field strength, and $E$ = modulus of elasticity.
        \column{0.55\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=0.9\linewidth]{fig/lec02/mag_3.pdf}
                \caption{Explanation of magnetostriction by the change in length (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
    \end{columns}
\end{frame}

\begin{frame}{Magnetostrictive Effect: Principle of Operation (2)}
    \begin{columns}
        \column{0.45\textwidth}
            \textbf{Physical Description:}
            \begin{itemize}
                \item The effect arises from the rotation of \textbf{Weiss domains} aligning with the external magnetic field.
                \item The material volume remains constant — only shape changes.
                \item Increase in length → \textit{positive magnetostriction}; decrease → \textit{negative magnetostriction}.
                \item Typical change: $-3\times10^{-5}$ m (Ni) to $+5\times10^{-5}$ m (Fe).
            \end{itemize}
        \column{0.55\textwidth}

\end{frame}

\begin{frame}{Inductive Effects: Principle of Operation (1)}

            \textbf{Key Inductive Phenomena in Sensors:}
            \begin{itemize}

\begin{frame}{Magnetic Field and Coil Inductance}
    \small
    \textbf{Magnetic Field in a Coil:}
    \begin{equation}
        B = \mu_0 \mu_r \frac{NI}{\ell}
    \end{equation}
    where $\mu_0$ = magnetic constant, $\mu_r$ = relative permeability, $N$ = number of turns, $I$ = current, and $\ell$ = length of coil.\\
    \medskip
    \textbf{Flux Linkage and Inductance:}
    \begin{equation}
        \Phi = B A = \mu_0 \mu_r \frac{N I A}{\ell}, \quad L = \frac{N \Phi}{I} = \mu_0 \mu_r \frac{N^2 A}{\ell}
    \end{equation}
    \textbf{Magnetic Resistance:}
    \begin{equation}
        R_{\text{magn}} = \frac{\ell_m}{\mu_0 \mu_r A}
    \end{equation}
    Inverse of permeability — analogous to resistance in an electrical circuit.
\end{frame}

\begin{frame}{Inductive Speed and Rotation Measurement}
    \small
    \textbf{Principle:}  
    A rotating ferromagnetic toothed wheel modulates magnetic flux through a stationary coil or permanent magnet.  
    The changing flux induces a voltage.
    \begin{columns}
        \column{0.52\textwidth}
            \medskip
            \textbf{Flux Variation:} $B_0$ -- DC component + $\hat{B}$-- AC component, Z: number of teeth
            \begin{equation}
                B = B_0 + \hat{B} \sin(Z \omega t)
            \end{equation}
            \textbf{Induced Voltage:} for effective cross sectional area $A$
            \begin{equation}
                u_{\text{ind}} = N \hat{B} A Z \omega \cos(Z \omega t)
            \end{equation}
            \textbf{Key Relationships:} $K_G$ -- sensor/geometric constant
            \begin{equation}
                u_S = K_G (B \sin(\omega t)) \frac{d\varphi}{dt}, \quad \hat{u}_S = 2 \pi K_G B n
            \end{equation}
            \vspace{-0.25cm}
            \begin{itemize}
                \item Output voltage amplitude $\propto$ rotational speed.
                \item Used in speed, frequency, and position detection.
            \end{itemize}

\begin{frame}{Capacitor and Capacitance: Basic Principle}
    \textbf{Definition:}
    A \textbf{capacitor} consists of two mutually insulated, oppositely charged conductor surfaces.  
    When a voltage $U$ is applied, charge $Q$ accumulates on the plates, defined by:
    \begin{equation}
        C = \frac{Q}{U}
    \end{equation}
    The unit of capacitance is the \textbf{Farad (F)}.
    \begin{columns}
        \column{0.55\textwidth}

                \item $d$ - distance between plates

\begin{frame}{Capacitance in the AC Circuit}

    \begin{equation}
        X_C = -j \frac{d}{\omega \varepsilon_0 \varepsilon_r A}

    \textbf{Interpretation:}
    \begin{itemize}
        \item Reactance $X_C$ changes proportionally with plate distance $d$.
        \item Thus, capacitance sensors can detect distance or approach of objects.
    \end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Gaussian Effect: Principle of Operation (1) %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Gaussian Effect: Principle of Operation (1)}
    \textbf{Definition:}
    The \textbf{Gaussian Effect} is a \textit{galvanomagnetic effect} that occurs when a current-carrying conductor or semiconductor is placed in a magnetic field $\mathbf{B}$ that is \textbf{not parallel} to the current flow.\\
    \begin{columns}
        \column{0.48\textwidth}
            \textbf{Physical basis:}
            \begin{itemize}
                \item A \textbf{Lorentz force} $\mathbf{F_L}$ acts on moving charge carriers.
                \item This deflects electrons, causing them to follow a \textbf{zigzag path}.
                \item The effective path length of electrons increases → \textbf{increased resistance}.
            \end{itemize}
        \column{0.52\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=1\linewidth]{fig/lec02/gauss_1.pdf}
                \caption{Origin of Gaussian effect: electron deflection in magnetic field (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
    \end{columns}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Gaussian Effect: Principle of Operation (2) %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Gaussian Effect: Principle of Operation (2)}
    \textbf{Lorentz Force:}
    \begin{equation}
        \mathbf{F_L} = q(\mathbf{E} + \mathbf{v} \times \mathbf{B})
    \end{equation}
    \textbf{Ohm's Law:} \quad $\mathbf{j} = \sigma \mathbf{E}$,  
    where $\mathbf{j}$ is current density and $\sigma$ is electrical conductivity.
    \begin{figure}
        \centering
        \includegraphics[width=0.8\linewidth]{fig/lec02/gauss_1.pdf}
        \caption{Origin of Gaussian effect: electron deflection in magnetic field (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
    \end{figure}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Microscopic Description of the Gaussian Effect %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Microscopic Description of the Gaussian Effect}
    \small
    \textbf{Microscopic relations:}
    \begin{equation}
        \sigma = q n \mu
    \end{equation}
    where:
    $q$ - charge of carrier, $n$ - carrier density, $\mu$ - mobility of charge carriers.\\
    \textbf{Current density:}
    \begin{equation}
        \mathbf{j} = \rho \mathbf{v}, \quad \rho = q n
    \end{equation}
    \textbf{Drift velocity:}
    \begin{equation}
        \mathbf{v} = \mu \mathbf{E}
    \end{equation}
    \textbf{Force balance:}
    \begin{equation}
        \mathbf{F} = q \mathbf{E} \quad \Rightarrow \quad \mathbf{v} = \frac{\mu \mathbf{F}}{q}
    \end{equation}
    \textbf{Interpretation:}
    Electron drift depends on both $\mathbf{E}$ and $\mathbf{B}$, leading to altered trajectories and measurable resistance variation.
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Derivation of Conductivity Change in Magnetic Field %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Derivation of Conductivity Change in Magnetic Field}
    When $\mathbf{B}$ is perpendicular to the electron motion:
    \begin{columns}
        \column{0.6\textwidth}
            \begin{equation}
                \mathbf{v} =
                \begin{pmatrix}
                v_x \\ v_y
                \end{pmatrix}
                = \mu
                \begin{pmatrix}
                E + v_y B \\
                - v_x B
                \end{pmatrix}
            \end{equation}
            Solving for $v_x$:
            \begin{equation}
                v_x = \frac{\mu E}{1 + \mu^2 B^2}
            \end{equation}
            \textbf{Implication:}
            \begin{itemize}
                \item Drift velocity \textbf{decreases} as $B$ increases.
                \item Current density $j = \rho v_x$ also decreases.
                \item Therefore, conductivity $\sigma$ decreases, and resistance increases.
            \end{itemize}
            \textbf{Resulting resistance relation:}
            \begin{equation}
                R(B) = R_0 \left(1 + \mu^2 B^2\right)
            \end{equation}
        \column{0.4\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=0.7\linewidth]{fig/lec02/gauss_2.pdf}
                \caption{Variation of resistance with magnetic flux density due to Gaussian effect (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
            \vspace{-0.25cm}
            \textbf{Conclusion:} Resistance increases quadratically with magnetic flux density.
    \end{columns}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Application of the Gaussian Effect %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Application of the Gaussian Effect}
    \small
    \textbf{1. Magnetic Field Measurement:}
    \begin{itemize}
        \item Used in \textbf{Magneto-Resistive Sensors (MDRs)} and field plates.
        \item Resistance $R$ varies with $\mathbf{B}$ → can be calibrated for field strength.
    \end{itemize}
    \textbf{2. Field Plate Construction:}
    \begin{itemize}
        \item A ceramic substrate with a thin semiconductor layer (e.g., InSb).
        \item Embedded metallic needles (e.g., NiSb) guide current paths.
        \item Magnetic flux density $B$ perpendicular to plate alters resistance.
    \end{itemize}
    \textbf{3. Typical properties of field plates (Table 2.13):}
    \begin{itemize}
        \item Output resistance $R_0$: 10 $\Omega$ - 10 k$\Omega$.
        \item Temperature coefficient $\alpha \approx -0.004\,\text{K}^{-1}$.
        \item Max operating temperature $\approx 95^\circ$C.
        \item Max power $\approx$ 0.5 W.
    \end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Working Mechanism Summary %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Working Mechanism Summary}
    \small
    \textbf{Electron motion:}
    \begin{itemize}
        \item Without magnetic field → straight-line conduction.
        \item With magnetic field → Lorentz force causes zigzag trajectory.
    \end{itemize}
    \textbf{Effect on resistance:}
    \begin{itemize}
        \item Longer electron paths increase collision frequency.
        \item Effective resistivity rises proportionally to $B^2$.
    \end{itemize}
    \textbf{Materials used:}
    \begin{itemize}
        \item High-mobility semiconductors: InSb, InAs, GaAs, Si.
        \item Often doped with Te to adjust resistive response.
        \item Typically exhibit \textbf{negative temperature coefficients.}
    \end{itemize}
\end{frame}

    \textbf{Key outcomes:}

\subsection{Hall Effect}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Hall Effect: Principle of Operation %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Hall Effect: Principle of Operation}
    \small
    The \textbf{Hall Effect}, like the Gaussian Effect, belongs to the \textit{galvanomagnetic effects}.
    \textbf{Concept:}
    \begin{itemize}
        \item When a current-carrying conductor or semiconductor is placed in a \textbf{magnetic field} $\mathbf{B}$ with a perpendicular component, charge carriers experience a \textbf{Lorentz force}.
    \end{itemize}
    \begin{columns}
        \column{0.48\textwidth}
            \vspace{-0.5cm}
            \begin{itemize}
                \item This deflects the electrons, creating a potential difference across the conductor — the \textbf{Hall voltage} $U_H$.
            \end{itemize}
            \textbf{Lorentz force:}
            \vspace{-0.5cm}
            \begin{equation}
                F_L = -q v_x B_z
            \end{equation}
            \textbf{Electric counterforce at equilibrium:}
            \begin{equation}
                F_{el} = -q E_y \quad \text{and} \quad -q E_y = -q v_x B_z
            \end{equation}
            \textbf{Therefore:}
            \begin{equation}
                E_y = v_x B_z
            \end{equation}
            \begin{equation}
                U_H = E_y b = v_x B_z b
            \end{equation}
        \column{0.48\textwidth}
            \begin{figure}
                \centering
                \vspace{-1cm}
                \includegraphics[width=0.75\linewidth]{fig/lec02/hall_1.pdf}
                \caption{Hall effect: charge carrier deflection and Hall voltage generation (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
    \end{columns}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Mathematical Description of Hall Voltage %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Mathematical Description of Hall Voltage}
    \small
    \textbf{Current density relation:}
    \begin{equation}
        j_x = n e v_x
    \end{equation}
    where:
    \begin{itemize}
        \item $n$ – number of charge carriers per unit volume,
        \item $e$ – elementary charge,
        \item $v_x$ – carrier velocity in $x$-direction.
    \end{itemize}
    \textbf{Hall voltage:}
    \begin{equation}
        U_H = \frac{1}{n e} j_x B_z b
    \end{equation}
    Defining the \textbf{Hall coefficient:}
    \begin{equation}
        A_H = \frac{1}{n e}
    \end{equation}
    Then:
    \begin{equation}
        U_H = A_H j_x B_z b
    \end{equation}
    \textbf{Hall resistance:}
    \begin{equation}
        R_H = \frac{A_H B_z}{d} \quad \Rightarrow \quad U_H = R_H I_x
    \end{equation}
    \textbf{Interpretation:} $U_H$ increases linearly with $B_z$ and decreases with carrier density $n$.
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Physical Interpretation %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Physical Interpretation}
    \small
    \textbf{Key points:}
    \begin{itemize}
        \item A magnetic field deflects charge carriers (electrons or holes) perpendicular to the current.
        \item The resulting charge separation builds up a Hall voltage $U_H$.
        \item At equilibrium, the Lorentz force equals the electric force.
    \end{itemize}
    \textbf{Dependence:}
    \begin{itemize}
        \item $U_H \propto I_x B_z$
        \item $U_H \propto 1 / (n e)$ — inversely proportional to carrier density.
        \item Independent of the specific resistivity of the material.
    \end{itemize}
    \textbf{Implications:}
    \begin{itemize}
        \item Semiconductors produce a much larger $U_H$ than metals.
        \item Sign of $U_H$ indicates whether conduction is via electrons or holes.
    \end{itemize}
\end{frame}

        \item The Hall voltage is proportional to $I_x B_z$.

\subsection{Thermoelectric Effect}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% \begin{frame}
% \textbf{\Large{Thermoelectric Effect}}
% \end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Thermoelectric Effect -- Principle %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Thermoelectric Effect -- Principle}
    \small
    \textbf{Definition:}
    When two dissimilar conductors are joined at both ends and exposed to different temperatures
    ($T_{AB}$, $T_{BA}$), a voltage $U_{th}$ is generated due to charge carrier diffusion.
    \begin{equation}
        \Delta T = T_{BA} - T_{AB}
    \end{equation}
    This phenomenon is known as the \textbf{Seebeck Effect}.
    \textbf{Physical explanation:}
    \begin{itemize}
        \item Thermal energy of a metal is stored partly as atomic oscillation and partly as electron kinetic energy.
        \item Electrons at the hotter end gain higher kinetic energy and diffuse toward the colder end.
        \item This diffusion creates a potential difference — the \textbf{thermoelectric voltage}.
    \end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Seebeck Coefficient and Its Origin %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Seebeck Coefficient and Its Origin}
    \small
    \textbf{Theoretical Seebeck coefficient of material A:}
    \begin{equation}
        S'_A = -\frac{1}{3e} \frac{d}{dT} \left\langle \frac{1}{2} m v^2 \right\rangle
    \end{equation}
    where:
    \begin{itemize}
        \item $e$ – electron charge,
        \item $T$ – absolute temperature,
        \item $\langle \frac{1}{2} m v^2 \rangle$ – average kinetic energy of conduction electrons.
    \end{itemize}
    \textbf{Simplified expression (for metals):}
    \begin{equation}
        S'_A = -\frac{\pi^2 k_B^2 T}{2 e E_F}
    \end{equation}
    where $k_B$ is the Boltzmann constant and $E_F$ the Fermi energy.
    \textbf{Interpretation:}
    \begin{itemize}
        \item $S'_A$ is material-dependent.
        \item It is positive for \textit{p-type} materials (hole conduction) and negative for \textit{n-type}.
    \end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Thermoelectric Voltage Relation %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Thermoelectric Voltage Relation}
    \small
    When conductors A and B are connected at two points with different temperatures $T_{AB}$ and $T_{BA}$:
    \begin{equation}
        U_{th} = \int_{T_{AB}}^{T_{BA}} S_{AB}(T) \, dT
    \end{equation}
    where:
    \begin{equation}
        S_{AB}(T) = S_A(T) - S_B(T)
    \end{equation}
    is the \textbf{differential Seebeck coefficient} between materials A and B.
    \textbf{Key features:}
    \begin{itemize}
        \item $U_{th}$ is proportional to the temperature difference $\Delta T$.
        \item $S_{AB}$ depends on material pair and temperature.
        \item Platinum is often used as the reference material ($S=0$).
    \end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Material Dependence of the Seebeck Effect %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Material Dependence of the Seebeck Effect}
    \small
    \textbf{Typical thermoelectric voltages (relative to platinum, 0–100 °C):}
    \begin{itemize}
        \item Platinum: 0.0 mV
        \item Constantan: –3.2 mV
        \item Nickel: –1.9 mV
        \item Tungsten: +0.7 mV
        \item Copper: +0.7 mV
        \item Iron: +1.9 mV
        \item Nickel–chrome: +2.2 mV
    \end{itemize}
    \textbf{Interpretation:}
    \begin{itemize}
        \item The sign indicates current direction (electron or hole conduction).
        \item Combining two materials (e.g., Cu–Constantan) forms a thermocouple with measurable $U_{th}$.
        \item For Cu–Constantan: $0.7 + 3.2 = 3.9$ mV for $\Delta T = 100$ °C.
    \end{itemize}
\end{frame}

    \begin{itemize}
        \item Simple, robust, no external power supply required.
        \item Wide temperature range, fast response, miniaturizable.
    \end{itemize}
\end{frame}

\subsection{Thermoresistance Effect}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% \end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Thermoresistance Effect — Principle %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Thermoresistance Effect — Principle}
    \small
    \textbf{Concept:}
    The electrical resistance of materials depends on:
    \begin{itemize}
        \item Composition and crystal structure,
        \item Degree of purity and homogeneity,
        \item Temperature $T$.
    \end{itemize}
    For metals and metal alloys, resistance increases with temperature:
    \begin{equation}
        R(T) = R_0 \left[1 + a(T - T_0) + b(T - T_0)^2 \right]
    \end{equation}
    where:
    \begin{itemize}
        \item $R_0$ - resistance at reference temperature $T_0$ (0 °C or 20 °C),
        \item $a$ - temperature coefficient (linear term),
        \item $b$ - higher-order coefficient (non-linear term).
    \end{itemize}
    \textbf{Mechanism:} Electron scattering increases with atomic vibrations and impurities, raising resistance.
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Temperature Coefficient and Platinum Sensors %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Temperature Coefficient and Platinum Sensors}
    \small
    \textbf{Temperature coefficient:}
    \begin{equation}
        a = \frac{1}{R_0}\frac{dR(T)}{dT}
    \end{equation}
    Typical values:
    \begin{itemize}
        \item Platinum (Pt): $3.89 \times 10^{-3}\,\mathrm{K^{-1}}$
        \item Nickel-Chromium: $(0.02 - 0.05) \times 10^{-3}\,\mathrm{K^{-1}}$
    \end{itemize}
    \textbf{Platinum sensors (Pt 100, Pt 1000):}
    \begin{itemize}
        \item Pt 100 → 100 $\Omega$ at 0 °C; Pt 1000 → 1000 $\Omega$ at 0 °C.
        \item Highly stable, chemically resistant, and accurate.
        \item Range: –220 °C to +1000 °C (depending on type).
    \end{itemize}
    \textbf{Equation fit:} Linear up to 100 °C; slight deviation above.
\end{frame}

%% Thermal Response and Self-Heating %%

\begin{frame}{Thermal Response and Self-Heating}
    \small
    \textbf{Thermal response time:}
    \begin{itemize}
        \item Time for sensor resistance to change by 50 \% ($t_{0.5}$) or 90 \% ($t_{0.9}$) of a temperature step.
        \item Depends on medium (air, water) and sensor design.
    \end{itemize}
    \textbf{Self-heating effect:}
    \begin{equation}
        \Delta T = P \cdot S, \quad P = I^2 R
    \end{equation}
    where $S$ is the self-heating coefficient (K/mW).  
    → Higher measuring current increases sensor temperature and error.
    \textbf{Measuring currents (Heraeus recommendation):}
    \begin{itemize}
        \item Pt 100: 0.3-1.0 mA, Pt 1000: 0.1-0.3 mA.
    \end{itemize}
    \textbf{Applications:}
    \begin{itemize}
        \item Automotive, HVAC, industrial process, and life-science temperature sensing.
        \item High-accuracy measurement (–196 °C to +1000 °C).
    \end{itemize}
\end{frame}

\subsection{Photoelectric Effect}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Photoelectric Effect - Overview (1) %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Photoelectric Effect - Overview (1)}
    \textbf{Definition:}
    The photoelectric effect occurs when a \textbf{photon (quantum of light)} is absorbed by an \textbf{electron}, releasing it from its bond.\\
    \textbf{Key Principle:}
    \begin{itemize}
        \item A photon transfers its energy to an electron.
        \item The photon's energy must exceed the electron’s \textbf{binding energy}.
    \end{itemize}
    \textbf{Types of Photoelectric Effect:}
    \begin{itemize}
        \item \textbf{Outer photoelectric effect:} Electron emission from metal surfaces.
        \item \textbf{Inner photoelectric effect:} Electron excitation within semiconductors.
        \item \textbf{Photoionization:} Complete removal of an electron from atoms or molecules.
    \end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Photoelectric Effect - Overview (2) %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Photoelectric Effect – Overview (2)}
    \begin{figure}
        \centering
        \includegraphics[width=0.75\linewidth]{fig/lec02/photoelec_1.pdf}
        \caption{Types of photoelectric effect: outer, inner, and photoionization (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
    \end{figure}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Outer Photoelectric Effect – Principle %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Outer Photoelectric Effect – Principle}
    \begin{columns}
        \column{0.5\textwidth}
            \textbf{Mechanism:}
            \begin{itemize}
                \item Short-wave electromagnetic radiation (light) strikes a metal surface.
                \item Electrons are ejected — creating a measurable \textbf{photocurrent}.
            \end{itemize}
            \textbf{Energy relationship:}
            \begin{equation}
                E_{\text{kin}} = h f - W_A
            \end{equation}
            where
            \begin{itemize}
                \item $h$ – Planck’s constant ($6.626 \times 10^{-34}\,\mathrm{Js}$)
                \item $f$ – frequency of incident light
                \item $W_A$ – work function (binding energy)
            \end{itemize}
        \column{0.5\textwidth}
            \vspace{-0.5cm}
            \begin{figure}
                \centering
                \includegraphics[width=0.9\linewidth]{fig/lec02/photoelec_2.pdf}
                \caption{Outer photoelectric effect: photon absorption and electron emission (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
            \hspace{-0.5cm}
            \textbf{Key Implications:}
            \begin{itemize}
                \item No electrons emitted if $h f < W_A$
                \item \textbf{Cut-off frequency:} $f_{\text{gr}} = W_A / h$
                \item Increasing light intensity increases photocurrent but not electron energy.
            \end{itemize}
    \end{columns}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Outer Photoelectric Effect – Characteristics %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Outer Photoelectric Effect – Characteristics}
    \begin{columns}
        \column{0.6\textwidth}
            \textbf{Observations:}
            \begin{itemize}
                \item The \textbf{kinetic energy} of photoelectrons depends only on light \textbf{frequency}, not intensity.
                \item The \textbf{photocurrent} increases with light intensity.
                \item There exists a \textbf{threshold frequency} below which no emission occurs.
            \end{itemize}
            \textbf{Photon Energy Relation:}
            \begin{equation}
                E_{\text{ph}} = h f = \frac{h c}{\lambda}
            \end{equation}
            Thus:
            \begin{equation}
                E_{\text{ph}} \propto \frac{1}{\lambda}
            \end{equation}
        \column{0.4\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=0.75\linewidth]{fig/lec02/photoelec_2b.pdf}
                \caption{Outer photoelectric effect: photon absorption and electron emission (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
    \end{columns}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Inner Photoelectric Effect – Photoconductivity %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Inner Photoelectric Effect – Photoconductivity}
    \textbf{Mechanism:}
    \begin{itemize}
        \item In semiconductors, photons lift electrons from the \textbf{valence band} to the \textbf{conduction band}.
        \item This increases the number of free charge carriers → higher conductivity.
    \end{itemize}
    \begin{columns}
        \column{0.6\textwidth}
            \textbf{Energy Condition:}
            \begin{equation}
                E_{\text{ph}} = h f = \frac{h c}{\lambda} \geq E_g
            \end{equation}
            or equivalently:
            \begin{equation}
                \lambda \leq 1.24\,\mu\text{m·eV} / E_g
            \end{equation}
            \textbf{Band Model Explanation:}
            \begin{itemize}
                \item Photons with $E_{\text{ph}} > E_g$ generate electron-hole pairs.
                \item More photons → higher conductivity (photoconduction).
            \end{itemize}
        \column{0.4\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=1\linewidth]{fig/lec02/photoelec_3.pdf}
                \caption{Inner photoelectric effect: photon absorption and electron excitation in semiconductors (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
    \end{columns}
\end{frame}

%% Applications of Photoelectric Effect %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Applications of Photoelectric Effect}
    \textbf{Practical Uses:}
    \begin{itemize}
        \item \textbf{Photon detectors} (photomultipliers, photodiodes)
        \item \textbf{Optocouplers} for galvanic isolation
        \item \textbf{Light barriers and light curtains} for automation
        \item \textbf{Light and color sensors} for measurement and control
        \item \textbf{Photovoltaic cells} (solar energy conversion)
    \end{itemize}
    \textbf{Summary:}
    \begin{itemize}
        \item Outer effect → metal emission and photocurrent.
        \item Inner effect → semiconductor excitation and increased conductivity.
    \end{itemize}
\end{frame}

%% Refraction and Reflection of Light %%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Refraction and Reflection of Light}
    \small
    \textbf{Refraction:}
    When light passes from one medium to another, its speed and direction change.
    \begin{equation}
        \frac{\sin \varepsilon}{\sin \beta} = \frac{c}{c'} = \frac{n'}{n}
    \end{equation}
    where $n$ = refractive index, $c$ = speed of light in vacuum, $c'$ = in medium.
    \begin{columns}
        \column{0.5\textwidth}
            \textbf{Snell's Law:}
            \begin{equation}n = \frac{c_0}{c}
            \end{equation}
            \textbf{Reflection:}
            \begin{itemize}
                \item Angle of incidence = angle of reflection.
                \item Total internal reflection occurs if angle of incidence $> \text{critical angle}$.
            \end{itemize}
            \textbf{Applications:}
            \begin{itemize}
                \item Optical fibers utilize total internal reflection to guide light efficiently.
            \end{itemize}
        \column{0.5\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=0.5\linewidth]{fig/lec02/opto_2.pdf}
                \caption{Refraction and reflection of light at an interface (adapted from \href{https://doi.org/10.1007/978-3-658-34920-2}{Hering \& Schönfelder})}
            \end{figure}
    \end{columns}
\end{frame}

\begin{frame}{Optical Sensors – Principle of Operation}
    \small
    \textbf{Basic Components:}
    \begin{itemize}
        \item \textbf{Transmitter:} light source (e.g., LED, laser).
        \item \textbf{Transmission medium:} air, glass, or fiber.
        \item \textbf{Receiver:} converts light signal into electrical signal (e.g., photodiode).
    \end{itemize}
    \textbf{Working Concept:}
    \begin{itemize}
        \item Based on absorption, reflection, or refraction of light.
        \item Used for object detection, distance sensing, and imaging.
    \end{itemize}
    \textbf{Equation of photon energy:}
    \begin{equation}
        E_{\text{ph}} = h \cdot f = \frac{h \cdot c}{\lambda}
    \end{equation}
    \textbf{Examples:}
    \begin{itemize}
        \item Photoelectric sensors, optical encoders, light barriers, and camera modules.
    \end{itemize}
\end{frame}

    \begin{figure}

    \textbf{Photon-Energy Interaction:}
    \begin{itemize}
        \item \textbf{Absorption:} photon excites electron from $E_1 \rightarrow E_2$.

        \item \textbf{Stimulated emission:} incident photon induces another photon of same phase → amplification.
    \end{itemize}
\end{frame}
